cmake_minimum_required(VERSION 3.8)
project(agv_nav2_plugins)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_CXX_STANDARD 17)
add_compile_options(-Wall -Wextra -O2)
find_package(ament_cmake REQUIRED)
set(DEPS rclcpp rclcpp_lifecycle pluginlib nav2_core nav2_costmap_2d nav2_util nav2_behaviors nav2_msgs
  nav_msgs geometry_msgs sensor_msgs std_msgs tf2 tf2_ros)
foreach(d ${DEPS})
  find_package(${d} REQUIRED)
endforeach()
include_directories(include)
add_library(${PROJECT_NAME} SHARED
  src/route_planner.cpp src/route_controller.cpp src/goal_checker.cpp src/adjust_pose.cpp)
ament_target_dependencies(${PROJECT_NAME} ${DEPS})
pluginlib_export_plugin_description_file(nav2_core plugins.xml)
install(TARGETS ${PROJECT_NAME} LIBRARY DESTINATION lib ARCHIVE DESTINATION lib RUNTIME DESTINATION bin)
install(DIRECTORY include/ DESTINATION include)
install(FILES plugins.xml DESTINATION share/${PROJECT_NAME})
# 几何内核单测 (不依赖 ROS): colcon build 后运行 build/agv_nav2_plugins/test_geom
add_executable(test_geom test/test_geom.cpp)
ament_package()
