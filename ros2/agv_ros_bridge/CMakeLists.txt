cmake_minimum_required(VERSION 3.8)
project(agv_ros_bridge)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_CXX_STANDARD 17)
add_compile_options(-Wall -Wextra -O2)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(builtin_interfaces REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(rclcpp_action REQUIRED)
find_package(nav2_msgs REQUIRED)
find_package(std_msgs REQUIRED)
# 规划算法 C 库 (planning/native/agvnav.c: 车体净空/拐点过弯，自研导引用) 与精定位头文件 (agv_nav2_plugins)
set(AGVNAV_SRC ${CMAKE_CURRENT_SOURCE_DIR}/../../planning/native/agvnav.c)
set_source_files_properties(${AGVNAV_SRC} PROPERTIES COMPILE_FLAGS "-ffp-contract=off")
enable_language(C)
add_executable(agv_ros_bridge src/bridge_node.cpp ${AGVNAV_SRC})
target_include_directories(agv_ros_bridge PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../agv_nav2_plugins/include)
ament_target_dependencies(agv_ros_bridge rclcpp rclcpp_action nav_msgs nav2_msgs std_msgs sensor_msgs geometry_msgs builtin_interfaces tf2_ros)
install(TARGETS agv_ros_bridge DESTINATION lib/${PROJECT_NAME})
ament_package()
