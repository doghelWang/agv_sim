cmake_minimum_required(VERSION 3.8)
project(agv_ros_bridge)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_CXX_STANDARD 17)
add_compile_options(-Wall -Wextra -O2)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(builtin_interfaces REQUIRED)
find_package(tf2_ros REQUIRED)
add_executable(agv_ros_bridge src/bridge_node.cpp)
ament_target_dependencies(agv_ros_bridge rclcpp nav_msgs sensor_msgs geometry_msgs builtin_interfaces tf2_ros)
install(TARGETS agv_ros_bridge DESTINATION lib/${PROJECT_NAME})
ament_package()
